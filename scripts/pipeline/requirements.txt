# SSI Pipeline — Dependencies
# Core pipeline now requires numpy for vectorized 10k Monte Carlo per
# Phase 1 PR-1 (AUDIT_v4_0_2 §7.7 Decision 1: F-L3-1 + F-L3-3 + F-L3-4).
# See PHASE_1_IMPLEMENTATION_PLAN.md PR-1.

numpy>=1.24,<2.0    # Vectorized Monte Carlo (canonical scoring engine, PR-2)

# ── Phase 1.5 data ingestion (P15-A, 8 June 2026) ──
# cdsapi unblocks Copernicus CDS ERA5 + CMIP6 fetching for the 28 countries
# missing climate baselines after Phase 1's --all refresh. Requires operator's
# CDS API key — register free at https://cds.climate.copernicus.eu, then
# either export CDS_API_KEY=<key> or place it in ~/.cdsapirc per the cdsapi
# package convention. See scripts/pipeline/data/README.md §"Climate".
cdsapi>=0.7.0       # Copernicus Climate Data Store API client
netCDF4>=1.6.0      # NetCDF reader for ERA5 .nc files returned by CDS
                    # (uncommented during P15-A triage 8 June 2026 — Korea
                    # ERA5 fetch hit "netCDF4 not available" after CDS
                    # download succeeded; install unblocks the parse step)
rasterio>=1.3       # GeoTIFF reader for GEM 2023.1 Global Seismic Hazard
                    # Map raster (P15-B-2, 8 June 2026 — switched from
                    # imagined CSV to actual GeoTIFF format GEM publishes).
                    # Heavy dep (~50 MB with GDAL backend) but only used by
                    # the seismic.py global-fallback path. Operators who
                    # pre-extract their own CSV at gem_global_pga475.csv
                    # don't need rasterio installed.

# ── Cross-border polygon enforcement (Discipline #36, 18 June 2026) ──
# shapely powers the point-in-polygon check in scripts/check_cross_border.py
# + scripts/pipeline/utils/geo.py {load_country_polygon, is_inside_country,
# filter_by_country_polygon, cross_border_audit}. Lazy-imported so existing
# helpers (haversine_km, nearest_grid_value, bilinear_interpolate,
# load_substations) keep working without shapely if the polygon helpers
# aren't called. The 18 Jun 2026 audit found Austria 47.5% / Canada 74.4% /
# Greenland 86.5% of substations were outside their declared national
# polygon — this dep unblocks the structural fix. See
# CROSS_BORDER_SUBSTATION_AUDIT_20260618.md.
shapely>=2.0        # Polygon ops + topology healing via buffer(0)
